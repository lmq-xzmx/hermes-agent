#### 9.6.2 PKCE 实现（必须）

```python
# services/oauth_service.py - PKCE 支持
import hashlib
import secrets
import base64

class PKCEService:
    """PKCE (Proof Key for Code Exchange) 实现"""

    @staticmethod
    def generate_code_verifier() -> str:
        """生成 code_verifier (43-128字符)"""
        return secrets.token_urlsafe(64)[:128]

    @staticmethod
    def generate_code_challenge(code_verifier: str) -> str:
        """生成 code_challenge (SHA256 + Base64url)"""
        digest = hashlib.sha256(code_verifier.encode()).digest()
        return base64.urlsafe_b64encode(digest).rstrip(b'=').decode()

    @staticmethod
    def generate_state() -> str:
        """生成 state 参数防止 CSRF"""
        return secrets.token_urlsafe(32)

    def validate_code_challenge(self, code_verifier: str, code_challenge: str, method: str = 'S256') -> bool:
        """验证 code_challenge"""
        if method == 'S256':
            expected = self.generate_code_challenge(code_verifier)
            return secrets.compare_digest(expected, code_challenge)
        return False
```

#### 9.6.3 Token 刷新机制

```python
class TokenRefreshService:
    """Token 刷新服务"""

    def __init__(self, db_pool):
        self.db_pool = db_pool

    async def refresh_access_token(self, refresh_token: str) -> dict:
        """刷新 Access Token"""
        # 1. 验证 Refresh Token
        payload = jwt.decode(refresh_token, JWT_SECRET, algorithms=['HS256'])

        if payload.get('type') != 'refresh':
            raise AuthError('Invalid token type')

        # 2. 检查 Token 是否已撤销
        jti = payload.get('jti')
        if await self._is_token_revoked(jti):
            raise AuthError('Token has been revoked')

        # 3. 生成新的 Access Token
        new_access_token = self._generate_access_token(payload['sub'])

        # 4. 轮转 Refresh Token（单次使用）
        await self._revoke_token(jti)
        new_refresh_token = self._generate_refresh_token(payload['sub'])

        return {
            'access_token': new_access_token,
            'refresh_token': new_refresh_token,
            'token_type': 'bearer',
            'expires_in': ACCESS_TOKEN_EXPIRE_MINUTES * 60
        }

    async def revoke_token(self, token: str) -> None:
        """撤销 Token（登出）"""
        payload = jwt.decode(token, JWT_SECRET, algorithms=['HS256'])
        jti = payload.get('jti')
        await self._revoke_token(jti)
```
